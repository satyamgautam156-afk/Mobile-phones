# ── List dikhao ──────────────────────────────
def show_mobiles():
    print("\n--- Mobiles ---")
    print(f"  {'No':<4} {'Model':<30} {'Price'}")
    print("  " + "-" * 45)
    for i, (name, price) in enumerate(mobiles.items(), 1):
        print(f"  {i:<4} {name:<30} Rs. {price:,}")
    print("  " + "-" * 45)

def show_accessories():

# ── Bill banana ──────────────────────────────
def make_bill():
    import datetime
    now = datetime.datetime.now().strftime("%d-%m-%Y  %I:%M %p")

    print(f"\n{'='*48}")
    print(f"       {STORE_NAME}")
    print(f"       {STORE_ADDRESS}")
    print(f"       Tel: {STORE_PHONE}")
    print(f"{'='*48}")

    customer = input("Customer naam (Enter skip): ").strip()
    phone    = input("Phone number (Enter skip):  ").strip()
    imei     = input("IMEI Number (Enter skip):   ").strip()

    bill = []

    while True:
        print("\n--- What to add ? ---")
        print("  1. Mobile")
        print("  2. Accessory")
        print("  0. Bill Complete")
        choice = input("Choose: ").strip()

        if choice == "0":
            break

        elif choice == "1":
            show_mobiles()
            num = input("Add Mobile number : ").strip()
            if not num.isdigit() or int(num) < 1 or int(num) > len(mobiles):
                print("❌ oops sorry wrong number!")
                continue
            name, price = list(mobiles.items())[int(num)-1]
            qty_input = input(f"Quantity: ").strip()
            if not qty_input.isdigit() or int(qty_input) < 1:
                print("❌ created quantity !")
                continue
            qty   = int(qty_input)
            total = qty * price
            bill.append(("📱 " + name, qty, price, total))
            print(f"✅ Added: {name} x{qty} = Rs. {total:,}")

        elif choice == "2":
            show_accessories()
            num = input("Accessory number : ").strip()
            if not num.isdigit() or int(num) < 1 or int(num) > len(accessories):
                print("❌ oops wrong number!")
                continue
            name, price = list(accessories.items())[int(num)-1]
            qty_input = input(f"Quantity: ").strip()
            if not qty_input.isdigit() or int(qty_input) < 1:
                print("❌  created quantity !")
                continue
            qty   = int(qty_input)
            total = qty * price
            bill.append(("🔌 " + name, qty, price, total))
            print(f"✅ Added: {name} x{qty} = Rs. {total:,}")

        else:
            print("❌ 0, 1 ya 2 type!")

    if not bill:
        print("\nBill empty!")
        return

    # Discount
    disc_input = input("\nDiscount (Rs) — 0 = null: ").strip()
    discount   = int(disc_input) if disc_input.isdigit() else 0

    subtotal    = sum(t for _, _, _, t in bill)
    grand_total = subtotal - discount

    # ── Bill Print ──
    print(f"\n{'='*48}")
    print(f"        {STORE_NAME}")
    print(f"        {STORE_ADDRESS}")
    print(f"        Tel: {STORE_PHONE}")
    print(f"{'='*48}")
    print(f"Date    : {now}")
    if customer: print(f"Customer: {customer}")
    if phone:    print(f"Phone   : {phone}")
    if imei:     print(f"IMEI    : {imei}")
    print(f"{'-'*48}")
    print(f"{'Item':<28} {'Qty':>3} {'Price':>8} {'Total':>8}")
    print(f"{'-'*48}")

    for name, qty, price, total in bill:
        print(f"{name:<28} {qty:>3} {price:>8,} {total:>8,}")

    print(f"{'-'*48}")
    print(f"{'Subtotal':>40}: Rs. {subtotal:,}")
    if discount:
        print(f"{'Discount':>40}: Rs. {discount:,}")
    print(f"{'TOTAL':>40}: Rs. {grand_total:,}")
    print(f"{'='*48}")
    print("     Thank you! ")
    print(f"{'='*48}\n")

    # Save
    save = input("Bill save karein? (y/n): ").strip().lower()
    if save == "y":
        fname = f"mobile_bill_{datetime.datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"
        with open(fname, "w", encoding="utf-8") as f:
            f.write(f"{STORE_NAME}\n{STORE_ADDRESS}\nTel: {STORE_PHONE}\n")
            f.write(f"Date: {now}\n")
            if customer: f.write(f"Customer: {customer}\n")
            if phone:    f.write(f"Phone: {phone}\n")
            if imei:     f.write(f"IMEI: {imei}\n")
            f.write("-" * 48 + "\n")
            for name, qty, price, total in bill:
                f.write(f"{name:<28} x{qty}  Rs.{total:,}\n")
            f.write("-" * 48 + "\n")
            f.write(f"Subtotal : Rs. {subtotal:,}\n")
            if discount:
                f.write(f"Discount : Rs. {discount:,}\n")
            f.write(f"TOTAL    : Rs. {grand_total:,}\n")
            f.write("Shukriya! \n")
        print(f"✅ Saved: {fname}")
